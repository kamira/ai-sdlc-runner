#!/bin/bash
# Run a command with only the variables a child `claude`/`codex` needs. Inheriting this session's
# CLAUDE_* variables makes every child append to this session's env file (codex plugin hook), which
# grows until every shell fails with E2BIG.
args=()
for v in HOME PATH TERM LANG USER HTTPS_PROXY HTTP_PROXY NO_PROXY https_proxy http_proxy no_proxy \
         SSL_CERT_FILE SSL_CERT_DIR NODE_EXTRA_CA_CERTS REQUESTS_CA_BUNDLE CURL_CA_BUNDLE \
         ANTHROPIC_BASE_URL CODEX_HOME PYTHONPATH; do
  [ -n "${!v+x}" ] && args+=("$v=${!v}")
done
exec env -i "${args[@]}" "$@"
